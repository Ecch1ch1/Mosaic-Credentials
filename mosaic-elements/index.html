How to Use in Any Web Page
<!-- 1. Include the drop-in component -->
<script src="mosaic-credential-input.js"></script>

<!-- 2. Use it inside your form just like an <input> -->
<form action="/login" method="POST">
  <input type="text" name="username" required />
  <mosaic-credential-input name="password" placeholder="Enter secret..."></mosaic-credential-input>
  <button type="submit">Sign In</button>
</form>
